const express = require("express"); const mongoose = require("mongoose"); const cors = require("cors"); require("dotenv").config(); const app = express(); app.use(cors()); app.use(express.json()); /* ========================= 🔌 CONNEXION MONGO ========================= */ mongoose.connect(process.env.MONGO_URI) .then(() => console.log("✅ MongoDB connecté")) .catch(err => console.log("❌ Mongo error", err)); /* ========================= 📦 MODELS ========================= */ const Proprietaire = mongoose.model("Proprietaire", new mongoose.Schema({ nom: String, telephone: String, tantiemes: Number, appartement: String })); const Charge = mongoose.model("Charge", new mongoose.Schema({ libelle: String, montantTotal: Number, date: { type: Date, default: Date.now } })); /* ========================= 🌱 SEED (auto init) ========================= */ app.get("/seed", async (req, res) => { await Proprietaire.deleteMany(); await Charge.deleteMany(); await Proprietaire.insertMany([ { nom: "Mamadou Ndiaye", telephone: "221776348031", tantiemes: 150, appartement: "A101" }, { nom: "Fatou Diop", telephone: "221782617843", tantiemes: 250, appartement: "B202" }, { nom: "Ibrahima Fall", telephone: "221781112233", tantiemes: 600, appartement: "C303" } ]); const charge = await Charge.create({ libelle: "Charges Nettoyage & Sécurité", montantTotal: 500000 }); res.json({ message: "Seed OK", chargeId: charge._id }); }); /* ========================= 📊 STATS ========================= */ app.get("/api/stats", async (req, res) => { const nb = await Proprietaire.countDocuments(); const total = await Charge.aggregate([ { $group: { _id: null, total: { $sum: "$montantTotal" } } } ]); res.json({ nbProprietaires: nb, totalCharges: total[0]?.total || 0, nbPaiements: 18, nbImpayes: 6 }); }); /* ========================= 💰 APPELS DE FONDS ========================= */ app.get("/api/appels/:chargeId", async (req, res) => { const charge = await Charge.findById(req.params.chargeId); const props = await Proprietaire.find(); const total = props.reduce((s, p) => s + p.tantiemes, 0); const result = props.map(p => { const montant = (charge.montantTotal * p.tantiemes) / total; const msg = `Bonjour ${p.nom}, appel de fonds: ${montant.toFixed(0)} FCFA`; const wa = `https://wa.me/${p.telephone}?text=${encodeURIComponent(msg)}`; return { nom: p.nom, appartement: p.appartement, montant, whatsapp: wa }; }); res.json(result); }); /* ========================= 🌐 FRONTEND (HTML DIRECT) ========================= */ app.get("/", (req, res) => { res.send(` PAV Syndic

🏢 PAV Syndic Dashboard

`); }); /* ========================= 🚀 SERVER START ========================= */ const PORT = process.env.PORT || 5000; app.listen(PORT, () => console.log("🚀 Server running on " + PORT));
